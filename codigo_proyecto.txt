export const authGuard: CanActivateFn = async () => {
  const auth = inject(AuthService);
  const router = inject(Router);

    await auth.esperarInicializacion();


  if (auth.estaAutenticado()) {
    return true;
  }
  return router.parseUrl('/auth/login');
};

export const roleGuard = (rolesPermitidos: RolUsuario[]): CanActivateFn => {
  return () => {
    const auth = inject(AuthService);
    const router = inject(Router);
    const rolActual = auth.rol();

    if (rolActual && rolesPermitidos.includes(rolActual)) {
      return true;
    }

    // Si no tiene el rol, redirige a inicio
    return router.parseUrl('/');
  };
};

import { Injectable, signal, computed, inject } from '@angular/core';
import { Router } from '@angular/router';
import { SupabaseService } from '../supabase.service';
import { Perfil, RolUsuario } from '../models/database.types';

export interface RegistroDTO {
  email: string;
  password: string;
  nombre: string;
  apellido: string;
  fecha_nacimiento: string; // YYYY-MM-DD
  tipo_sangre: 'A+' | 'A-' | 'B+' | 'B-' | 'AB+' | 'AB-' | 'O+' | 'O-';

@Injectable({
  providedIn: 'root'
})
export class AuthService {

  private readonly supabaseService = inject(SupabaseService);

  // Signals de estado de sesión
  private perfilSignal = signal<Perfil | null>(null);
  private cargandoSignal = signal<boolean>(true);

  // Computeds públicos
  readonly perfil = computed(() => this.perfilSignal());
  readonly estaAutenticado = computed(() => !!this.perfilSignal());
  readonly rol = computed<RolUsuario | null>(() => this.perfilSignal()?.rol ?? null);
  readonly esAdmin = computed(() => this.perfilSignal()?.rol === 'admin');
  readonly esEmpleado = computed(() => this.perfilSignal()?.rol === 'empleado');
  readonly esCliente = computed(() => this.perfilSignal()?.rol === 'cliente');
  readonly cargando = computed(() => this.cargandoSignal());

  private inicializacion: Promise<void>;

  constructor(private router: Router) {
     this.inicializacion = this.inicializarSesion();
  }

  async esperarInicializacion(): Promise<void> {
  return this.inicializacion;
}

  private async inicializarSesion(): Promise<void> {
    try {
      const { data: { session } } = await this.supabaseService.client.auth.getSession();
      
      if (session?.user) {
        await this.cargarPerfil(session.user.id);
      }
    } finally {
      this.cargandoSignal.set(false);
    }

    // Escucha cambios en el estado de autenticación (login, logout, refresh)
    this.supabaseService.client.auth.onAuthStateChange(async (event, session) => {
      if (session?.user) {
        await this.cargarPerfil(session.user.id);
      } else {
        this.perfilSignal.set(null);
      }
    });
  }

  async cargarPerfil(userId: string): Promise<Perfil | null> {
    const { data, error } = await this.supabaseService.client
      .from('profiles')
      .select('*')
      .eq('id', userId)
      .single();

    if (error || !data) {
      console.error('Error cargando perfil:', error);
      this.perfilSignal.set(null);
      return null;
    }

    const perfil = data as Perfil;
    this.perfilSignal.set(perfil);
    return perfil;
  }

  /**
   * RF-AUT-01: Registro con los 7 campos requeridos por el cliente (01/01).
   * Nota de defensa oral: No usamos triggers en SQL. El servicio crea la cuenta
   * en Supabase Auth y luego inserta el registro en public.profiles en secuencia.
   */
  async registrar(dto: RegistroDTO): Promise<void> {
    // 1. Crear usuario en Auth
    const { data: authData, error: authError } = await this.supabaseService.client.auth.signUp({
      email: dto.email,
      password: dto.password
    });

    if (authError || !authData.user) {
      throw new Error(authError?.message || 'Error al crear credenciales de acceso.');
    }

    // 2. Insertar perfil en public.profiles (RN: rol por defecto es 'cliente')
    const { error: profileError } = await this.supabaseService.client
      .from('profiles')
      .insert({
        id: authData.user.id,
        email: dto.email,
        nombre: dto.nombre,
        apellido: dto.apellido,
        fecha_nacimiento: dto.fecha_nacimiento,
        tipo_sangre: dto.tipo_sangre,
        color_ojos: dto.color_ojos,
        dias_vacaciones: dto.dias_vacaciones,
        rol: 'cliente'
      });

    if (profileError) {
      throw new Error(`Error al guardar perfil de usuario: ${profileError.message}`);
    }

    await this.cargarPerfil(authData.user.id);
  }

  /**
   * RF-AUT-02: Inicio de sesión con email y contraseña.
   */
  async iniciarSesion(email: string, pass: string): Promise<Perfil> {
    const { data, error } = await this.supabaseService.client.auth.signInWithPassword({
      email,
      password: pass
    });

    if (error || !data.user) {
      throw new Error(error?.message || 'Credenciales inválidas.');
    }

    const perfil = await this.cargarPerfil(data.user.id);
    if (!perfil) throw new Error('No se encontró el perfil del usuario.');
    return perfil;
  }

  /**
   * Cierre de sesión.
   */
  async cerrarSesion(): Promise<void> {
    await this.supabaseService.client.auth.signOut();
    this.perfilSignal.set(null);
  }

}

import { inject, Injectable } from '@angular/core';
import { SupabaseClient } from '@supabase/supabase-js';
import { Butaca, Funcion, Sala } from '../../models/database.types';
import { SupabaseService } from '../../supabase.service';

@Injectable({
  providedIn: 'root'
})
export class FuncionesService {
  private supabase = inject(SupabaseService);
  private readonly MINUTOS_LIMPIEZA = 30; // RN-04: margen mínimo de 30 minutos

  /**
   * RN-04 / RN-05: Chequea si un rango de horarios colisiona con funciones existentes en una sala.
   * Considera la ventana de limpieza de 30 minutos antes y después.
   * Se valida en TypeScript/Angular y no con exclusion constraints de Postgres.
   */
  async hayConflicto(salaId: string, inicio: Date, fin: Date, funcionIdExcluida?: string): Promise<boolean> {
    // Margen de 30 min aplicado a la búsqueda
    const inicioConMargen = new Date(inicio.getTime() - this.MINUTOS_LIMPIEZA * 60 * 1000).toISOString();
    const finConMargen = new Date(fin.getTime() + this.MINUTOS_LIMPIEZA * 60 * 1000).toISOString();

    let query = this.supabase.client
      .from('funciones')
      .select('id, inicio, fin')
      .eq('sala_id', salaId)
      .neq('estado', 'cancelada')
      // Condición de solapamiento: inicio < finConMargen AND fin > inicioConMargen
      .lt('inicio', finConMargen)
      .gt('fin', inicioConMargen);

    if (funcionIdExcluida) {
      query = query.neq('id', funcionIdExcluida);
    }

    const { data: solapadas, error } = await query;
    if (error) {
      throw new Error(`Error al verificar conflictos de sala: ${error.message}`);
    }

    return (solapadas && solapadas.length > 0);
  }

  /**
   * RN-06: Asignación automática de sala.
   * Recorre las salas activas y prueba cada una con hayConflicto() hasta encontrar la primera libre.
   */
  async asignarSalaAutomatica(inicio: Date, fin: Date): Promise<Sala> {
    const { data: salas, error } = await this.supabase.client
      .from('salas')
      .select('*')
      .eq('activa', true)
      .order('numero', { ascending: true });

    if (error || !salas || salas.length === 0) {
      throw new Error('No hay salas activas disponibles en el sistema.');
    }

    for (const sala of salas) {
      const enConflicto = await this.hayConflicto(sala.id, inicio, fin);
      if (!enConflicto) {
        return sala; // Sala asignada con éxito
      }
    }

    throw new Error('No hay ninguna sala disponible que respete la ventana de 30 minutos para el horario solicitado.');
  }

  /**
   * RF-ADM-04 / RF-ADM-05: Da de alta una función calculando su horario de finalización
   * en base a la duración de la película y asignando sala automáticamente si no se provee.
   */
  async crearFuncionAutomatica(params: {
    peliculaId: string;
    inicio: Date;
    formato: '2D' | '3D' | '4D' | '5D';
    idioma: 'castellano' | 'subtitulada';
    precioBase: number;
    userId: string;
  }): Promise<Funcion> {
    // 1. Obtener duración de la película
    const { data: pelicula, error: errPeli } = await this.supabase.client
      .from('peliculas')
      .select('duracion_min')
      .eq('id', params.peliculaId)
      .single();

    if (errPeli || !pelicula) {
      throw new Error('Película no encontrada.');
    }

    // Calcular fin en TS antes de insertar 
    const fin = new Date(params.inicio.getTime() + pelicula.duracion_min * 60 * 1000);

    // 2. Asignación automática de sala (RN-06)
    const salaAsignada = await this.asignarSalaAutomatica(params.inicio, fin);

    // 3. Insertar la función
    const { data: nuevaFuncion, error: errInsert } = await this.supabase.client
      .from('funciones')
      .insert({
        pelicula_id: params.peliculaId,
        sala_id: salaAsignada.id,
        inicio: params.inicio.toISOString(),
        fin: fin.toISOString(),
        formato: params.formato,
        idioma: params.idioma,
        precio_base: params.precioBase,
        estado: 'programada',
        created_by: params.userId
      })
      .select()
      .single();

    if (errInsert) {
      throw new Error(`Error al persistir la función: ${errInsert.message}`);
    }

    /// Me guardo el log de auditoria para ver quien hace que
    await this.supabase.client.from('audit_log').insert({ accion: 'crear_funcion', entidad: 'funciones', entidad_id: nuevaFuncion.id, actor_id: params.userId })

    return nuevaFuncion;
  }

  async getButacasDeFuncion(funcionId: string): Promise<{ salaId: string, butacas: Butaca[] }> {
    // 1. Obtener la sala
    const { data: funcion, error: errFuncion } = await this.supabase.client
      .from('funciones')
      .select('sala_id')
      .eq('id', funcionId)
      .single();

    if (errFuncion || !funcion) throw new Error("Función no encontrada");

    // 2. Traer las butacas de esa sala
    const { data: butacas, error: errButacas } = await this.supabase.client
      .from('butacas')
      .select('*')
      .eq('sala_id', funcion.sala_id)
      .order('numero', { ascending: true });

    if (errButacas) throw new Error("Error al cargar las butacas");

    return {
      salaId: funcion.sala_id,
      butacas: butacas as Butaca[]
    };
  }

  async getNombrePelicula(funcionId: string) {
    const { data, error } = await this.supabase.client
      .from('funciones')
      .select(`
      pelicula_id,
      peliculas (
        titulo,
        poster_url,
        duracion_min,
        clasificacion
      )
    `)
      .eq('id', funcionId)
      .single();

    if (error || !data) {
      throw new Error('Función no encontrada');
    }
    return data.peliculas;
  }

  async getFuncion(funcionId: string) {
    const { data: funcion, error: errFuncion } = await this.supabase.client
      .from('funciones')
      .select(`
        *,
        salas(
          nombre
        )`)
      .eq('id', funcionId)
      .single();

    if (errFuncion || !funcion) throw new Error("Función no encontrada");
    return funcion;
  }

}

import { Injectable, inject } from '@angular/core';
import { SupabaseService } from '../../core/supabase.service';

export interface Cupon {
  id?: string;
  codigo: string;
  descripcion?: string | null;
  tipo: 'primera_compra' | 'edad_minima' | 'general';
  porcentaje: number;
  edad_minima: number | null;   // sólo tiene sentido cuando tipo = 'edad_minima'
  activo: boolean;
}

export interface Combo {
  id?: string;
  nombre: string;
  descripcion?: string | null;
  precio: number;
  destacado?: boolean;          // RF-COM-10: debe poder destacarse en la compra
  activo: boolean;
}

@Injectable({
  providedIn: 'root'
})
export class AdminConfigService {

  // --- GESTIÓN DE CUPONES ---

  private supabase = inject(SupabaseService);
  
  async getCupones(): Promise<Cupon[]> {
    const { data, error } = await this.supabase.client.from('cupones').select('*').order('created_at', { ascending: false });
    if (error) throw new Error('Error al cargar los cupones');
    return data as Cupon[];
  }

  async guardarCupon(cupon: Cupon) {
    const { error } = await this.supabase.client.from('cupones').upsert({
      id: cupon.id,
      codigo: cupon.codigo.toUpperCase(),
      descripcion: cupon.descripcion ?? null,
      tipo: cupon.tipo,
      porcentaje: cupon.porcentaje,
      edad_minima: cupon.edad_minima,
      activo: cupon.activo,
    });
    
    if (error) throw new Error(`Error guardando el cupón: ${error.message}`);
  }

  async toggleEstadoCupon(id: string, estadoActual: boolean) {
    const { error } = await this.supabase.client.from('cupones').update({ activo: !estadoActual }).eq('id', id);
    if (error) throw new Error('Error al cambiar el estado del cupón');
  }

  // --- GESTIÓN DE COMBOS ---

  async getCombos(): Promise<Combo[]> {
    const { data, error } = await this.supabase.client.from('combos').select('*').order('created_at', { ascending: false });
    if (error) throw new Error('Error al cargar los combos');
    return data as Combo[];
  }

  async guardarCombo(combo: Combo) {
    const { error } = await this.supabase.client.from('combos').upsert({
      id: combo.id,
      nombre: combo.nombre,
      descripcion: combo.descripcion ?? null,
      precio: combo.precio,
      destacado: combo.destacado ?? true,
      activo: combo.activo,
    });

    if (error) throw new Error(`Error guardando el combo: ${error.message}`);
  }

  async toggleEstadoCombo(id: string, estadoActual: boolean) {
    const { error } = await this.supabase.client.from('combos').update({ activo: !estadoActual }).eq('id', id);
    if (error) throw new Error('Error al cambiar el estado del combo');
  }
}

import { Component, inject, OnInit, signal } from '@angular/core';
import { CommonModule } from '@angular/common';
import { AdminReportesService } from '../admin-reportes.service';
import jsPDF from 'jspdf';

@Component({
  imports: [CommonModule],
  selector: 'app-admin-dashboard',
  styleUrl: './admin-dashboard.component.css',
  templateUrl: './admin-dashboard.component.html',
})
export class AdminDashboardComponent implements OnInit {

  private reportesService = inject(AdminReportesService);

  cargando = signal(true);
  datos = signal<any>({ facturacionDiaria: 0, entradasVendidas: 0, topPeliculas: [], topCandy: [], auditoria: [] });

  ngOnInit() {
    this.cargarDatos();
  }

  async cargarDatos() {
    try {
      const data = await this.reportesService.getDashboardData();
      this.datos.set(data);
    } catch (error) {
      console.error("Error al cargar dashboard", error);
    } finally {
      this.cargando.set(false);
    }
  }

  calcularPorcentaje(vistas: number): number {
    const max = this.datos().topPeliculas[0]?.vistas || 1;
    return (vistas / max) * 100;
  }

  exportarExcel() {
    // Generar CSV nativo (reconocido como Excel) sin dependencias pesadas
    const d = this.datos();
    const headers = "Fecha Reporte,Facturacion Diaria,Entradas Vendidas\n";
    const row = `${new Date().toLocaleDateString()},${d.facturacionDiaria},${d.entradasVendidas}\n`;
    
    const blob = new Blob([headers + row], { type: 'text/csv;charset=utf-8;' });
    const link = document.createElement("a");
    const url = URL.createObjectURL(blob);
    link.setAttribute("href", url);
    link.setAttribute("download", `reporte_cine_${new Date().toISOString().split('T')[0]}.csv`);
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  }

  exportarPDF() {
    const doc = new jsPDF();
    const d = this.datos();
    
    doc.setFontSize(20);
    doc.text('Reporte de Dirección del Cine', 20, 20);
    
    doc.setFontSize(12);
    doc.text(`Fecha de emisión: ${new Date().toLocaleString()}`, 20, 30);
    
    doc.setFontSize(14);
    doc.text(`Resumen del Día`, 20, 50);
    doc.setFontSize(12);
    doc.text(`- Facturación Total: $${d.facturacionDiaria}`, 20, 60);
    doc.text(`- Entradas Vendidas: ${d.entradasVendidas} unidades`, 20, 70);
    
    doc.save(`reporte_admin_${new Date().toISOString().split('T')[0]}.pdf`);
  }

}

    this.procesando.set(true);
    this.mensajeError.set(null);
    this.resultadoExito.set(null);

    try {
      const peliculaSeleccionada = this.peliculas().find(p => p.id === this.formulario.peliculaId);
      if (!peliculaSeleccionada) throw new Error("Película inválida");

      const fechaInicio = new Date(this.formulario.fechaHora);

        this.formulario.formato,
        this.formulario.idioma,
        this.formulario.precioBase
      );

      this.resultadoExito.set(resultado.sala.nombre);

      // Limpiar formulario parcialmente
      this.formulario.fechaHora = '';

      this.mensajeError.set(e.message);

import { Injectable, inject } from '@angular/core';
import { SupabaseService } from '../../core/supabase.service';

@Injectable({
  providedIn: 'root'
})
export class AdminFuncionesService {

    private supabase = inject(SupabaseService);

  async getPeliculasActivas() {
    const { data, error } = await this.supabase.client.from('peliculas').select('id, titulo, duracion_min').eq('activa', true);
    if (error) throw error;
    return data;
  }

  /**
   * REGLA DE NEGOCIO: Asignación automática de sala sin superposiciones y con 30 min de limpieza.
   */
  async programarFuncion(peliculaId: string, duracionMinutos: number, fechaHoraInicio: Date, formato: string, idioma: string, precioBase: number) {
    // 1. Calcular el horario de fin de esta nueva función (Duración + 30 min de limpieza)
    const inicioMs = fechaHoraInicio.getTime();
    const finMs = inicioMs + (duracionMinutos * 60000) + (30 * 60000); 
    
    const inicioISO = new Date(inicioMs).toISOString();
    const finISO = new Date(finMs).toISOString();

    // 2. Traer todas las salas
    const { data: salas, error: errSalas } = await this.supabase.client.from('salas').select('id, nombre');
    if (errSalas || !salas) throw new Error('Error al cargar las salas del cine.');

    // 3. Buscar funciones que ocurran en ese mismo rango de tiempo para descartar salas
    // Buscamos funciones cuyo (inicio < nuevoFin) Y (fin_estimado > nuevoInicio)
    const { data: superposiciones, error: errSuperp } = await this.supabase.client
      .from('funciones')
      .select('sala_id')
      .lt('inicio', finISO)
      .gt('fin_estimado', inicioISO);

    if (errSuperp) throw new Error('Error al verificar disponibilidad de salas.');

    const salasOcupadas = new Set(superposiciones.map(f => f.sala_id));

    // 4. Encontrar la primera sala que NO esté en el Set de ocupadas
    const salaDisponible = salas.find(s => !salasOcupadas.has(s.id));

    if (!salaDisponible) {
      throw new Error('No hay ninguna sala disponible en ese horario. Por favor, elegí otra hora.');
    }

    // 5. Crear la función asignando la sala automáticamente
    const { data: nuevaFuncion, error: errInsert } = await this.supabase.client
      .from('funciones')
      .insert({
        pelicula_id: peliculaId,
        sala_id: salaDisponible.id,
        inicio: inicioISO,
        fin_estimado: finISO, // Guardamos el fin calculado (con los 30 min) para facilitar futuras queries
        formato: formato,
        idioma: idioma,
        precio_base: precioBase,
        estado: 'programada'
      })
      .select()
      .single();

    if (errInsert) throw new Error(`Error guardando la función: ${errInsert.message}`);

    return { funcion: nuevaFuncion, sala: salaDisponible };
  }
}

import { inject, Injectable } from '@angular/core';
import { SupabaseService } from '../../core/supabase.service';

@Injectable({
    providedIn: 'root'
})
export class AdminReportesService {

    private supabase = inject(SupabaseService);

    async getDashboardData() {
        // 1. Facturación Diaria y Entradas Vendidas
        const hoy = new Date();
        hoy.setHours(0, 0, 0, 0);
        const { data: ordenes, error: errOrdenes } = await this.supabase.client
            .from('ordenes')
            .select('total, entradas(id)')
            .eq('estado', 'pagada')
            .gte('pagada_at', hoy.toISOString());

        if (errOrdenes) throw new Error('Error al cargar la facturación diaria.');

        let facturacionDiaria = 0;
        let entradasVendidas = 0;

        ordenes?.forEach(orden => {
            facturacionDiaria += Number(orden.total);
            entradasVendidas += orden.entradas ? orden.entradas.length : 0;
        });

        // 2. Gráfico: Películas más vistas (Mes actual)
        const inicioMes = new Date(hoy.getFullYear(), hoy.getMonth(), 1).toISOString();
        const { data: entradasValidas } = await this.supabase.client
            .from('entradas')
            .select('funciones(peliculas(titulo))')
            .in('estado', ['valida', 'usada'])
            .gte('created_at', inicioMes);

        const conteoPeliculas: Record<string, number> = {};
        entradasValidas?.forEach((e: any) => {
            const titulo = e.funciones?.peliculas?.titulo;
            if (titulo) conteoPeliculas[titulo] = (conteoPeliculas[titulo] || 0) + 1;
        });

        const topPeliculas = Object.entries(conteoPeliculas)
            .map(([titulo, vistas]) => ({ titulo, vistas }))
            .sort((a, b) => b.vistas - a.vistas)
            .slice(0, 5);

        // 3. Producto de Candy Bar más vendido
        const { data: candyVendidos } = await this.supabase.client
            .from('orden_productos')
            .select('cantidad, productos(nombre)');

        const conteoCandy: Record<string, number> = {};
        candyVendidos?.forEach((c: any) => {
            const nombre = c.productos?.nombre;
            if (nombre) conteoCandy[nombre] = (conteoCandy[nombre] || 0) + c.cantidad;
        });

        const topCandy = Object.entries(conteoCandy)
            .map(([nombre, cantidad]) => ({ nombre, cantidad }))
            .sort((a, b) => b.cantidad - a.cantidad)
            .slice(0, 3);

        // 4. Log de Actividad Reciente
        const { data: auditoria } = await this.supabase.client
            .from('audit_log')
            .select('*, profiles(email)')
            .order('created_at', { ascending: false })
            .limit(15);

        return { facturacionDiaria, entradasVendidas, topPeliculas, topCandy, auditoria };
    }

  private readonly ahora = new Date();

  readonly peliculaCreada = output<void>();

  readonly errorMensaje = signal<string | null>(null);
  readonly exitoMensaje = signal<string | null>(null);
  readonly vistaPrevia = signal<string | null>(null);

  readonly peliculaForm = this.fb.nonNullable.group({

    sinopsis: ['', [Validators.required,Validators.maxLength(3000),Validators.pattern(/.*\S.*/)]],

    fecha_estreno: ['', Validators.required],
    clasificacion: ['atp' as Clasificacion, Validators.required],

  seleccionarArchivo(event: Event): void {
    const input = event.target as HTMLInputElement;
    const archivo = input.files?.[0];

    this.errorMensaje.set(null);
    this.exitoMensaje.set(null);

    if (!archivo) return;

    const tiposValidos = ['image/jpeg', 'image/png', 'image/webp'];

    if (!tiposValidos.includes(archivo.type)) {
      this.limpiarArchivo();
      input.value = '';
      this.errorMensaje.set('Formato no válido. Usá JPG, PNG o WebP.');
      return;
    }

    if (archivo.size > 5 * 1024 * 1024) {
      this.limpiarArchivo();
      input.value = '';
      this.errorMensaje.set('El póster no puede superar los 5 MB.');
      return;
    }

    this.limpiarArchivo();
    this.archivoSeleccionado = archivo;
    this.vistaPrevia.set(URL.createObjectURL(archivo));
  }

  limpiarArchivo(): void {
    const previa = this.vistaPrevia();

    if (previa) {
      URL.revokeObjectURL(previa);
    }

    this.archivoSeleccionado = null;
    this.vistaPrevia.set(null);
  }

  async onSubmit(): Promise<void> {
    this.errorMensaje.set(null);
    this.exitoMensaje.set(null);

    if (this.peliculaForm.invalid) {
      this.peliculaForm.markAllAsTouched();
      this.errorMensaje.set('Revisá los campos marcados antes de continuar.');
      return;
    }

    if (!this.archivoSeleccionado) {
      this.errorMensaje.set('Tenés que seleccionar el póster de la película.');
      return;
    }

    if (this.guardando()) return;

    this.guardando.set(true);

    try {
      const valores = this.peliculaForm.getRawValue();

      const dto: CrearPeliculaDTO = {
        titulo: valores.titulo.trim(),
        sinopsis: valores.sinopsis.trim(),
        duracion_min: valores.duracion_min,
        fecha_estreno: valores.fecha_estreno,
        clasificacion: valores.clasificacion,
        estado: valores.estado
      };

      await this.peliculasService.crear(dto, this.archivoSeleccionado);

      this.exitoMensaje.set('¡Película agregada al catálogo con éxito!');
      this.peliculaCreada.emit();

    } catch (error: unknown) {
      console.error('Error al guardar la película:', error);

      this.errorMensaje.set(
        error instanceof Error
          ? error.message
          : 'Ocurrió un error inesperado al guardar la película.'
      );
    } finally {
      this.guardando.set(false);
    }
  }

  ngOnDestroy(): void {
    this.limpiarArchivo();
  }

  ///SELECTOR FECHA PERSO

  onFechaCambio(fecha: Date): void {
  const fechaFormateada = [
    fecha.getFullYear(),
    String(fecha.getMonth() + 1).padStart(2, '0'),
    String(fecha.getDate()).padStart(2, '0'),
  ].join('-');

  this.peliculaForm.controls.fecha_estreno.setValue(fechaFormateada);
  this.peliculaForm.controls.fecha_estreno.markAsTouched();
}

}

import { inject, Injectable } from '@angular/core';
import { SupabaseService } from '../../../core/supabase.service';

export type Clasificacion = 'atp' | '13' | '18';

export type EstadoPelicula =
  | 'en_cartelera'
  | 'proximamente'
  | 'finalizado';

export interface CrearPeliculaDTO {
  titulo: string;
  sinopsis: string;
  duracion_min: number;
  clasificacion: Clasificacion;
  fecha_estreno: string;
  estado: EstadoPelicula;
}

@Injectable({
  providedIn: 'root'
})
export class PeliculasService {

    private readonly supabaseService = inject(SupabaseService);
  private readonly tabla = 'peliculas';
  private readonly bucket = 'posters';

  /**
   * Registra una película y sube su póster a Supabase Storage.
   */
  async crear(dto: CrearPeliculaDTO, poster: File): Promise<void> {
    // 1. Generar una ruta única para el póster.
    const extensionPorTipo: Record<string, string> = {
      'image/jpeg': 'jpg',
      'image/png': 'png',
      'image/webp': 'webp'
    };

    const extension = extensionPorTipo[poster.type];

    if (!extension) {
      throw new Error('El formato del póster no es válido.');
    }

    if (poster.size > 5 * 1024 * 1024) {
      throw new Error('El póster no puede superar los 5 MB.');
    }

    const rutaPoster = `${crypto.randomUUID()}.${extension}`;

    // 2. Subir la imagen.
    const { error: errorStorage } =
      await this.supabaseService.client.storage
        .from(this.bucket)
        .upload(rutaPoster, poster, {
          contentType: poster.type,
          upsert: false
        });

    if (errorStorage) {
      throw new Error(`Error al subir el póster: ${errorStorage.message}`);
    }

    try {
      // 3. Obtener la URL pública.
      const { data: imagen } =
        this.supabaseService.client.storage
          .from(this.bucket)
          .getPublicUrl(rutaPoster);

      // 4. Guardar la película.
      const { error: errorInsert } =
        await this.supabaseService.client
          .from(this.tabla)
          .insert({
            ...dto,
            poster_url: imagen.publicUrl
          });

      if (errorInsert) {
        throw new Error(
          `Error al registrar la película: ${errorInsert.message}`
        );
      }
    } catch (error: unknown) {
      // 5. Si falla el registro, intentar eliminar la imagen subida.
      const { error: errorEliminar } =
        await this.supabaseService.client.storage
          .from(this.bucket)
          .remove([rutaPoster]);

      if (errorEliminar) {
        console.error(
          'No se pudo eliminar el póster sobrante:',
          errorEliminar
        );
      }

      throw error;
    }
  }

}

export const fechaNoFuturaValidator: ValidatorFn = (
  control: AbstractControl
): ValidationErrors | null => {
  const valor = control.value;

  if (!valor) return null; // Validators.required se ocupa de los vacíos.

  const fecha = new Date(`${valor}T00:00:00`);
  const hoy = new Date();
  hoy.setHours(0, 0, 0, 0);

  return !Number.isNaN(fecha.getTime()) && fecha <= hoy
    ? null
    : { fechaFutura: true };
};

import { inject, Injectable } from '@angular/core';
import { Pelicula } from '../../core/models/database.types';
import { SupabaseService } from '../../core/supabase.service';

export interface PeliculaVentas extends Pelicula {
  entradas_vendidas: number;
}

@Injectable({
  providedIn: 'root'
})
export class CatalogoService {

  /**
   * Obtiene las películas marcadas por el admin para aparecer al entrar a la página.
   * RF-CAT-02: El cliente pidió poder elegir qué películas aparecen en la página principal[cite: 5].
   */

  private readonly supabaseService = inject(SupabaseService);

  async getPeliculasDestacadas(): Promise<Pelicula[]> {
    const { data, error } = await this.supabaseService.client
      .from('peliculas')
      .select('*')
      .eq('destacada', true)
      .eq('activa', true)
      .order('orden_destacada', { ascending: true });

    if (error) throw new Error(`Error al cargar destacadas: ${error.message}`);
    return data || [];
  }

  /**
   * Calcula y devuelve las 3 películas con más entradas vendidas.
   * RF-CAT-01: El cliente solicitó que se muestren primero las 3 películas más vendidas[cite: 5].
   * Nota técnica: Al no usar vistas SQL, traemos las entradas válidas y agrupamos en memoria.
   */
  async getTop3Vendidas(): Promise<PeliculaVentas[]> {
    // 1. Traer todas las películas activas
    const { data: peliculas, error: errPeliculas } = await this.supabaseService.client
      .from('peliculas')
      .select('*')
      .eq('activa', true);

    if (errPeliculas || !peliculas) throw new Error('Error al cargar catálogo para el top 3.');

    // 2. Traer las entradas válidas asociadas a sus funciones
    const { data: entradas, error: errEntradas } = await this.supabaseService.client
      .from('entradas')
      .select(`
        id,
        estado,
        funciones (
          pelicula_id
        )
      `)
      .in('estado', ['valida', 'usada']);

    if (errEntradas) throw new Error('Error al cargar ventas.');

    // 3. Agrupar y contar ventas por película en TypeScript
    const conteoVentas: Record<string, number> = {};
    peliculas.forEach(p => conteoVentas[p.id] = 0);

    entradas?.forEach((entrada: any) => {
      const peliculaId = entrada.funciones?.pelicula_id;
      if (peliculaId && conteoVentas[peliculaId] !== undefined) {
        conteoVentas[peliculaId]++;
      }
    });

    // 4. Mapear, ordenar y tomar las 3 primeras
    const peliculasConVentas: PeliculaVentas[] = peliculas.map(p => ({
      ...p,
      entradas_vendidas: conteoVentas[p.id]
    }));

    return peliculasConVentas
      .sort((a, b) => b.entradas_vendidas - a.entradas_vendidas)
      .slice(0, 3);
  }

  /**
   * Obtiene la lista de todos los géneros disponibles para los filtros.
   */
  async getGeneros() {
    const { data, error } = await this.supabaseService.client
      .from('generos')
      .select('*')
      .order('nombre', { ascending: true });

    if (error) throw new Error(`Error al cargar géneros: ${error.message}`);
    return data || [];
  }

  /**
   * Obtiene la lista preventas.
   */
  async getPeliculasPreventa(): Promise<Pelicula[]> {
    const { data, error } = await this.supabaseService.client
      .from('peliculas')
      .select('*')
      .eq('estado', 'proximamente')
      .eq('activa', true);

    if (error) throw new Error(`Error al cargar preventas: ${error.message}`);
    return data || [];
  }



  /**
   * RF-CAT-03 / RF-CAT-04: Búsqueda por texto (título/sinopsis) y filtro por múltiples géneros.
   * Nota de arquitectura: La consulta a Supabase trae las películas con sus géneros asociados 
   * mediante un join implícito (!inner) para poder filtrar correctamente.
   */
  async buscarPeliculas(termino: string, generosIds: string[]): Promise<Pelicula[]> {
    let query = this.supabaseService.client
      .from('peliculas')
      .select(`
        *,
        pelicula_generos!inner (
          genero_id
        )
      `)
      .eq('activa', true);

    // Filtro por texto en título o sinopsis
    if (termino && termino.trim() !== '') {
      query = query.or(`titulo.ilike.%${termino}%,sinopsis.ilike.%${termino}%`);
    }

    // Filtro por múltiples géneros (si seleccionó alguno)
    if (generosIds && generosIds.length > 0) {
      query = query.in('pelicula_generos.genero_id', generosIds);
    }

    const { data, error } = await query;

    if (error) throw new Error(`Error en la búsqueda: ${error.message}`);

    // Como una película con varios géneros puede venir duplicada en el join, filtramos únicos
    const unicas = Array.from(new Map((data || []).map((p: any) => [p.id, p])).values());

    return unicas as Pelicula[];
  }

  /**
   * RF-CAT-05: Ficha de película con géneros.
   */
  async getPeliculaById(id: string): Promise<Pelicula | null> {
    const { data, error } = await this.supabaseService.client
      .from('peliculas')
      .select(`
        *,
        pelicula_generos (
          generos ( nombre )
        )
      `)
      .eq('id', id)
      .single();


    if (error) throw new Error(`Error al cargar la película: ${error.message}`);
    const pelicula = data as any;

    return {
      ...pelicula,
      pelicula_generos: pelicula.pelicula_generos
        ?.map((pg: any) => pg.generos?.nombre)
        .filter(Boolean) ?? []
    };
  }

  /**
   * RF-RES-03: Calcula y muestra la puntuación promedio de cada película junto con la cantidad de reseñas.
   * Se agrupa en TS porque no usamos funciones SQL/vistas por requerimiento arquitectónico.
   */
  async getResumenResenas(peliculaId: string): Promise<{ promedio: number, total: number }> {
    const { data, error } = await this.supabaseService.client
      .from('resenias')
      .select('puntaje')
      .eq('pelicula_id', peliculaId);

    if (error) {
      console.error('Error cargando reseñas', error);
      return { promedio: 0, total: 0 };
    }

    const total = data.length;
    if (total === 0) return { promedio: 0, total: 0 };

    const suma = data.reduce((acc, curr) => acc + curr.puntaje, 0);
    return { promedio: Number((suma / total).toFixed(1)), total };
  }

  /**
   * RF-CAT-06: Lista las funciones disponibles indicando fecha, hora, sala, formato e idioma.
   */
  async getFuncionesDisponibles(peliculaId: string) {
    const ahora = new Date().toISOString();
    const { data, error } = await this.supabaseService.client
      .from('funciones')
      .select(`
        id, inicio, formato, idioma, precio_base, estado,
        salas ( nombre )
      `)
      .eq('pelicula_id', peliculaId)
      .eq('estado', 'programada')
      .gt('inicio', ahora) // Solo funciones futuras
      .order('inicio', { ascending: true });



    if (error) throw new Error(`Error al cargar funciones: ${error.message}`);
    return data;
  }

}

import { CommonModule } from '@angular/common';
import { Component, computed, inject, input, OnDestroy, OnInit, signal } from '@angular/core';
import { RouterLink } from '@angular/router';
import { CatalogoService } from '../../catalogo.service';

@Component({
  imports: [CommonModule, RouterLink],
  selector: 'app-pelicula-carrusel',
  styleUrl: './pelicula-carrusel.component.css',
  templateUrl: './pelicula-carrusel.component.html',
})
export class PeliculaCarruselComponent implements OnInit, OnDestroy {


   private catalogoService = inject(CatalogoService);
   resenas = signal<{ promedio: number, total: number }>({ promedio: 0, total: 0 });

  peliculas = input<any[]>([]);

  currentIndex = signal<number>(0);

  // Computed properties
  peliculaActual = computed(() => {
    const items = this.peliculas();
    return items.length > 0 ? items[this.currentIndex()] : null;
  });

  private slideInterval: any;

  async ngOnInit() {
    if (this.peliculas().length > 0) {
      this.iniciarAutoplay();
    }
    this.resenas.set( await this.catalogoService.getResumenResenas(this.peliculaActual().id));
  }

  ngOnDestroy() {
    this.limpiarIntervalo();
  }

  setSlide(index: number) {
    this.currentIndex.set(index);
    this.limpiarIntervalo();
    this.iniciarAutoplay();
  }

  private siguienteSlide() {
    const total = this.peliculas().length;
    if (total > 0) {
      this.currentIndex.update(i => (i + 1) % total);
    }
  }

  private iniciarAutoplay() {
    this.slideInterval = setInterval(() => {
      this.siguienteSlide();
    }, 6000);
  }

  private limpiarIntervalo() {
    if (this.slideInterval) {
      clearInterval(this.slideInterval);
    }
  }

}

import { Component, effect, inject, OnInit, signal } from '@angular/core';
import { CatalogoService } from '../catalogo.service';
import { Genero, Pelicula } from '../../../core/models/database.types';
import { RouterLink } from '@angular/router';
import { UpperCasePipe } from '@angular/common';

@Component({
  imports: [RouterLink, UpperCasePipe],
  selector: 'app-lista-catalogo.component',
  styleUrl: './lista-catalogo.component.css',
  templateUrl: './lista-catalogo.component.html',
})
export class ListaCatalogoComponent implements OnInit {

  private catalogoService = inject(CatalogoService);

  // Estado reactivo
  generos = signal<Genero[]>([]);
  peliculas = signal<Pelicula[]>([]);
  cargando = signal<boolean>(true);

  // Filtros
  terminoBusqueda = signal<string>('');
  seleccionados = signal<string[]>([]);

  constructor() {
    // Este effect() se dispara automáticamente cada vez que el usuario escribe o tilda un género
    effect(() => {
      const termino = this.terminoBusqueda();
      const generosIds = this.seleccionados();
      
      // Pequeño debounce rudimentario para no saturar Supabase al tipear rápido
      const timeoutId = setTimeout(() => {
        this.ejecutarBusqueda(termino, generosIds);
      }, 300);

      return () => clearTimeout(timeoutId);
    });
  }

  ngOnInit() {
    this.cargarGeneros();
  }

  async cargarGeneros() {
    try {
      const data = await this.catalogoService.getGeneros();
      this.generos.set(data);
    } catch (error) {
      console.error('Error cargando géneros', error);
    }
  }

  async ejecutarBusqueda(termino: string, generosIds: string[]) {
    this.cargando.set(true);
    try {
      const resultados = await this.catalogoService.buscarPeliculas(termino, generosIds);
      this.peliculas.set(resultados);
    } catch (error) {
      console.error('Error en búsqueda', error);
    } finally {
      this.cargando.set(false);
    }
  }

  toggleGenero(generoId: string) {
    const actuales = this.seleccionados();
    if (actuales.includes(generoId)) {
      this.seleccionados.set(actuales.filter(id => id !== generoId));
    } else {
      this.seleccionados.set([...actuales, generoId]);
    }
  }

  limpiarFiltros() {
    this.terminoBusqueda.set('');
    this.seleccionados.set([]);
  }

}

import { CommonModule } from '@angular/common';
import { Component, computed, inject, OnInit, signal } from '@angular/core';
import { ActivatedRoute, Router, RouterLink } from '@angular/router';
import { CatalogoService } from '../catalogo.service';
import { Pelicula } from '../../../core/models/database.types';

@Component({
  imports: [CommonModule],
  selector: 'app-pelicula-detalle.component',
  styleUrl: './pelicula-detalle.component.css',
  templateUrl: './pelicula-detalle.component.html',
})
export class PeliculaDetalleComponent implements OnInit {

  private route = inject(ActivatedRoute);
  private router = inject(Router);
  private catalogoService = inject(CatalogoService);

  pelicula = signal<Pelicula | null>(null);
  cargando = signal<boolean>(true);
  resenas = signal<{ promedio: number, total: number }>({ promedio: 0, total: 0 });

  // Array agrupado por día para iterar fácilmente en el HTML
  funcionesAgrupadas = signal<{ fecha: string, funciones: any[] }[]>([]);
  fechaSeleccionada = signal<string | null>(null);

  ngOnInit() {
    const id = this.route.snapshot.paramMap.get('id');
    if (id) {
      this.cargarDatos(id);
    }

  }
  fechaActiva = computed(() => {
  const grupos = this.funcionesAgrupadas();
  const seleccionada = this.fechaSeleccionada();
  
  return grupos.find(g => g.fecha === seleccionada)?.fecha
    ?? grupos[0]?.fecha
    ?? null;
  });

  funcionesFechaSeleccionada = computed(() => {
  const fecha = this.fechaActiva();

  return this.funcionesAgrupadas()
    .find(g => g.fecha === fecha)
    ?? null;
});


  async cargarDatos(id: string) {
    this.cargando.set(true);
    try {
      // Cargamos en paralelo para mayor velocidad
      const [peliculaData, resenasData, funcionesData] = await Promise.all([
        this.catalogoService.getPeliculaById(id),
        this.catalogoService.getResumenResenas(id),
        this.catalogoService.getFuncionesDisponibles(id)
      ]);

      this.pelicula.set(peliculaData);
      this.resenas.set(resenasData);
      this.agruparFuncionesPorDia(funcionesData || []);
    } catch (error) {
      console.error('Error cargando la ficha', error);
    } finally {
      this.cargando.set(false);
    }
  }

  private agruparFuncionesPorDia(funciones: any[]) {
    const agrupado = funciones.reduce((acc: any, funcion: any) => {
      // Extraemos solo la fecha (YYYY-MM-DD)
      const fecha = new Date(funcion.inicio).toISOString().split('T')[0];
      if (!acc[fecha]) acc[fecha] = [];
      acc[fecha].push(funcion);
      return acc;
    }, {});

    // Convertimos el objeto en array y ordenamos por fecha
    const arrayAgrupado = Object.keys(agrupado).map(fecha => ({
      fecha,
      funciones: agrupado[fecha]
    })).sort((a, b) => new Date(a.fecha).getTime() - new Date(b.fecha).getTime());

    this.funcionesAgrupadas.set(arrayAgrupado);
  }

  comprar(funcionId: string) {
    // Navegación al mapa de butacas
    this.router.navigate(['/compra/asientos', funcionId]);
  }

  seleccionarFecha(fecha: string) {
    this.fechaSeleccionada.set(fecha);
  }

}

  constructor() {

      butacasIds?: string[];
    };

    if (state.funcionId && state.butacasIds) {
      this.funcionId = state.funcionId;
      this.butacasIds = state.butacasIds;

      // No llegaron los datos necesarios
      this.router.navigate(['/']);
    }

    let vipCount = 0;
    let standardCount = 0; 

    const precioEstandar = this.funcion()?.precio_base; 
    const precioVip = precioEstandar * 0.5;

    butacas.forEach(butaca => {

      if (butaca.tipo === 'vip') {
        vipCount++;
      } else {
        standardCount++;
      }
    });

    return {
      vip: { cantidad: vipCount, subtotal: vipCount * precioVip },
      estandar: { cantidad: standardCount, subtotal: standardCount * precioEstandar }
    };
  });

  @HostListener('window:beforeunload')

    if (!this.compraFinalizadaExitosamente) {
      this.reservaService.limpiarMisReservas();
    }
  }

    }

    await this.cargarDatosDeCompraReales();
    this.pelicula.set(await this.funcionesService.getNombrePelicula(this.funcionId));
    this.funcion.set(await this.funcionesService.getFuncion(this.funcionId));
    
  }

      this.precioBase.set(datos.precioBase);
      this.butacasSeleccionadas.set(datos.butacas);

      this.mensajeError.set(error.message || "Error al cargar los detalles de la compra.");
    } finally {
      this.procesando.set(false);
    }
  }

    const prods = await this.comprasService.getProductosCandy();
    this.productosCandy.set(prods);
  }

  agregarCandy(prod: any) {
    const actual = this.carritoCandy();
    const existe = actual.find(c => c.productoId === prod.id);

    if (existe) {
      this.carritoCandy.set(actual.map(c =>
        c.productoId === prod.id ? { ...c, cantidad: c.cantidad + 1 } : c
      ));
    } else {
      this.carritoCandy.set([...actual, {
        productoId: prod.id,
        nombre: prod.nombre,
        cantidad: 1,
        precioUnitario: prod.precio
      }]);
    }
  }

  calcularTotal(): number {
    const subtotalEntradas = this.comprasService.calcularSubtotal(this.precioBase(), this.butacasSeleccionadas());
    const totalCandy = this.carritoCandy().reduce((acc, c) => acc + (c.cantidad * c.precioUnitario), 0);
    return subtotalEntradas + totalCandy;
  }

  async pagar() {
    if (!this.auth.estaAutenticado() && !this.emailInvitado) {
      this.mensajeError.set("Por favor ingresá un email para enviarte las entradas.");
      return;
    }

    this.procesando.set(true);
    this.mensajeError.set(null);

    try {

      const ordenId = await this.comprasService.confirmarCompra({
        funcionId: this.funcionId,
        butacas: this.butacasSeleccionadas(),
        precioBase: this.precioBase(),
        emailInvitado: this.emailInvitado,
        candy: this.carritoCandy(),
        totalPagado: this.calcularTotal()
      });

      // 2. Si todo sale bien (sin colisiones en la BD), redirigimos al PDF
      this.compraFinalizadaExitosamente.set(true);
      this.router.navigate(['/compra/exito', ordenId]);
    } catch (e: any) {
      this.mensajeError.set(e.message);
    } finally {
      this.procesando.set(false);
    }
  }

  ngOnDestroy() {
    const paginaAnterior = this.router.url.includes('compra/asientos');

    if (!this.compraFinalizadaExitosamente() && !paginaAnterior) {
      this.reservaService.limpiarMisReservas();
    }
  }



}

import { Injectable, inject } from '@angular/core';
import { AuthService } from '../../core/auth/auth.service';
import { Butaca } from '../../core/models/database.types';
import { SupabaseService } from '../../core/supabase.service';

export interface CarritoCandy {
    productoId?: string;
    comboId?: string;
    nombre: string;
    cantidad: number;
    precioUnitario: number;
}

@Injectable({
    providedIn: 'root'
})
export class ComprasService {
    private auth = inject(AuthService);
    private supabase = inject(SupabaseService);

    /**
     * RN-18: Calcula el subtotal desglosando recargos VIP.
     */
    calcularSubtotal(precioBase: number, butacas: Butaca[], recargoVipPct: number = 50): number {
        return butacas.reduce((total, b) => {
            let precioFila = precioBase;
            if (b.tipo === 'vip') {
                precioFila += precioBase * (recargoVipPct / 100);
            }
            return total + precioFila;
        }, 0);
    }

    /**
     * Valida la edad del comprador si la película tiene restricciones (RF-COM-07).
     */
    validarEdad(clasificacion: string, fechaNacimiento?: string): boolean {
        if (clasificacion === 'atp' || !fechaNacimiento) return true; // Si es anónimo asume responsabilidad (A-02)

        const edad = new Date().getFullYear() - new Date(fechaNacimiento).getFullYear();
        if (clasificacion === '13' && edad < 13) return false;
        if (clasificacion === '18' && edad < 18) return false;
        return true;
    }

    /**
     * Confirma la orden realizando los INSERTS secuenciales.
     * Genera los QR con crypto.randomUUID().
     */
    async confirmarCompra(params: {
        funcionId: string;
        butacas: Butaca[];
        precioBase: number;
        emailInvitado?: string;
        candy: CarritoCandy[];
        totalPagado: number;
    }) {

        const userId = this.auth.perfil()?.id ?? null;
        const emailFinal = userId ? this.auth.perfil()?.email : params.emailInvitado;

        if (!emailFinal) throw new Error("Se requiere un email para enviar las entradas.");

        try {
            // 1. Crear Orden
            const codigoCandy = params.candy.length > 0 ? crypto.randomUUID() : null;
            const { data: orden, error: errOrden } = await this.supabase.client
                .from('ordenes')
                .insert({
                    user_id: userId || null,
                    email_invitado: !userId ? emailFinal : null,
                    subtotal: params.totalPagado,
                    total: params.totalPagado,
                    estado: 'pagada', // Simulación de pago exitoso
                    pagada_at: new Date().toISOString(),
                    codigo_candy: codigoCandy,
                    candy_estado: codigoCandy ? 'pendiente' : null
                })
                .select()
                .single();

            if (errOrden) throw errOrden;

            // 2. Crear Entradas (con QR individual)
            const entradasToInsert = params.butacas.map(b => {
                let precio = params.precioBase;
                if (b.tipo === 'vip') precio += (params.precioBase * 0.5);

                return {
                    orden_id: orden.id,
                    funcion_id: params.funcionId,
                    butaca_id: b.id,
                    precio: precio,
                    tipo_butaca: b.tipo,
                    codigo_qr: crypto.randomUUID(), // QR único por entrada
                    estado: 'valida'
                };
            });

            const { error: errEntradas } = await this.supabase.client.from('entradas').insert(entradasToInsert);
            if (errEntradas) throw errEntradas;

            // 3. Crear items de Candy Bar (si hay)
            if (params.candy.length > 0) {
                const candyToInsert = params.candy.map(c => ({
                    orden_id: orden.id,
                    producto_id: c.productoId || null,
                    combo_id: c.comboId || null,
                    cantidad: c.cantidad,
                    precio_unitario: c.precioUnitario
                }));
                await this.supabase.client.from('orden_productos').insert(candyToInsert);
            }

            // 4. Limpiar reservas temporales
            const butacasIds = params.butacas.map(b => b.id);
            await this.supabase.client
                .from('reservas_butaca')
                .delete()
                .eq('funcion_id', params.funcionId)
                .in('butaca_id', butacasIds);

            // 5. Cargamos puntos

            const { error: errPuntos } = await this.supabase.client.from('puntos_movimientos').insert({
                user_id: userId,
                tipo: 'acumulacion',
                puntos: Math.floor(params.totalPagado), // 1 peso = 1 punto
                orden_id: orden.id,
                descripcion: `Puntos ganados por compra de orden #${orden.id}`,
            });
            if (errPuntos) {
                console.error("Error al guardar puntos: ", errPuntos);
            }

            return orden.id; // Retorna el ID para redirigir al PDF
        } catch (error: any) {
            throw new Error(`Fallo al confirmar la compra: ${error.message}`);
        }
    }

    // Métodos auxiliares para cargar catálogo de Candy Bar
    async getProductosCandy() {
        const { data } = await this.supabase.client.from('productos').select('*').eq('activo', true);
        return data || [];
    }

    /**
   * Obtiene la orden completa con sus entradas, funciones, salas, películas y productos de candy.
   * Usado para la generación del PDF de éxito.
   */
    async getOrdenCompleta(ordenId: string) {
        const { data, error } = await this.supabase.client
            .from('ordenes')
            .select(`
        *,
        entradas ( *, funciones ( inicio, pelicula_id, salas(nombre), peliculas(titulo, clasificacion) ) ),
        orden_productos ( cantidad, productos(nombre) )
      `)
            .eq('id', ordenId)
            .single();

        if (error) {
            throw new Error(`Error al recuperar la orden: ${error.message}`);
        }

        return data;
    }

    async getDatosCheckout(funcionId: string, butacasIds: string[]) {
        // Buscar precio de la función
        const { data: funcion, error: errFuncion } = await this.supabase.client
            .from('funciones')
            .select('precio_base')
            .eq('id', funcionId)
            .single();

        if (errFuncion || !funcion) {
            throw new Error("No se pudo cargar el precio de la función.");
        }

        // Buscar detalles de las butacas (para saber cuáles son VIP/estándar)
        const { data: butacas, error: errButacas } = await this.supabase.client
            .from('butacas')
            .select('*')
            .in('id', butacasIds);

        if (errButacas || !butacas) {
            throw new Error("No se pudieron cargar los detalles de las butacas.");
        }

        return {
            precioBase: funcion.precio_base,
            butacas: butacas as Butaca[]
        };
    }
}

import { Component, computed, HostListener, inject, OnDestroy, OnInit, signal } from '@angular/core';
import { ActivatedRoute, Router, RouterLink } from '@angular/router';
import { ReservasService } from '../reserva.service';
import { Butaca } from '../../../core/models/database.types';
import { CommonModule, CurrencyPipe } from '@angular/common';
import { FuncionesService } from '../../../core/services/funciones/funciones.service';
import { ComprasService } from '../compras.service';

@Component({
  imports: [CommonModule, CurrencyPipe],
  selector: 'app-mapa-butacas',
  styleUrl: './mapa-butacas.component.css',
  templateUrl: './mapa-butacas.component.html',
})
export class MapaButacasComponent implements OnInit, OnDestroy {
  private route = inject(ActivatedRoute);
  private router = inject(Router);
  private reservasService = inject(ReservasService);
  private funcionesService = inject(FuncionesService);
  private comprasService = inject(ComprasService);

  funcionId!: string;
  salaId!: string;
  pelicula: any;
  cargando = signal<boolean>(true);
  mensajeError = signal<string | null>(null);

  butacasFisicas = signal<Butaca[]>([]);
  precioEntradas = signal<number>(0);

  continuandoCompra = signal<boolean>(false);
  procesando = signal<string[]>([]); /// bloqueo optimista

  misSeleccionadas = computed(() => this.reservasService.misReservasActivas());

  // Computed que reconstruye la grilla de filas (A-T) con sus 3 columnas
  filasAgrupadas = computed(() => {
    const butacas = this.butacasFisicas();
    if (!butacas.length) return [];

    const agrupadas: Record<string, { letra: string, izquierda: Butaca[], centro: Butaca[], derecha: Butaca[] }> = {};

    butacas.forEach(b => {
      if (!agrupadas[b.fila]) agrupadas[b.fila] = { letra: b.fila, izquierda: [], centro: [], derecha: [] };
      if (b.columna === 'izquierda') agrupadas[b.fila].izquierda.push(b);
      if (b.columna === 'centro') agrupadas[b.fila].centro.push(b);
      if (b.columna === 'derecha') agrupadas[b.fila].derecha.push(b);
    });

    // Ordenar letras alfabéticamente
    return Object.values(agrupadas).sort((a, b) => a.letra.localeCompare(b.letra));
  });

  tieneVIP = computed(() => {
    const seleccionadas = this.misSeleccionadas();
    const butacas = this.butacasFisicas();
    return seleccionadas.some(id => {
      const b = butacas.find(x => x.id === id);
      // Las filas VIP definidas por negocio son R, S y T
      return b && ['R', 'S', 'T'].includes(b.fila);
    });
  });

  async actualizarPrecio() {
    const seleccionadas = this.misSeleccionadas();
    if (seleccionadas.length > 0) {
      const datos = await this.comprasService.getDatosCheckout(this.funcionId, seleccionadas);
      this.precioEntradas.set(this.comprasService.calcularSubtotal(datos.precioBase, datos.butacas));
    } else {
      this.precioEntradas.set(0);
    }

  }

  @HostListener('window:beforeunload')
  alCerrarPestana() {
    this.reservasService.limpiarMisReservas();
  }

  async ngOnInit() {
    this.funcionId = this.route.snapshot.paramMap.get('funcionId')!;
    if (this.funcionId) this.cargarMapa();
    this.pelicula = await this.funcionesService.getNombrePelicula(this.funcionId);
  }

  async cargarMapa() {
    try {
      // 1. El servicio se encarga de la lógica de BD
      const datos = await this.funcionesService.getButacasDeFuncion(this.funcionId);

      this.salaId = datos.salaId;
      this.butacasFisicas.set(datos.butacas);

      // 2. Conectar a Realtime
      await this.reservasService.inicializarTiempoReal(this.funcionId);
      // las ya seleccionadas
      if (this.misSeleccionadas().length > 0) {
        this.actualizarPrecio();
      }

    } catch (e: any) {
      this.mensajeError.set(e.message || "Error al cargar el mapa de butacas.");
    } finally {
      this.cargando.set(false);
    }
  }

  estadoButaca(butacaId: string): 'libre' | 'reservada' | 'ocupada' {
    const bloqueada = this.reservasService.butacasBloqueadas().find(b => b.id === butacaId);
    return bloqueada ? bloqueada.estado : 'libre';
  }

  async toggleButaca(butaca: Butaca) {

    if (this.procesando().includes(butaca.id)) return;

    this.mensajeError.set(null);

    this.procesando.update(p => [...p, butaca.id]);

    try {
      if (this.misSeleccionadas().includes(butaca.id)) {
        await this.reservasService.liberarButaca(this.funcionId, butaca.id);
      } else {
        if (this.misSeleccionadas().length >= 8) {
          this.mensajeError.set("Máximo 8 butacas por compra.");
          return;
        }

        const exito = await this.reservasService.reservarButaca(this.funcionId, butaca.id);
        if (!exito) {
          this.mensajeError.set("¡Alguien más acaba de seleccionar esta butaca! Elegí otra.");
        }
      }
      await this.actualizarPrecio();
      this.actualizarPrecio();
    } finally {
      this.procesando.update(p => p.filter(id => id !== butaca.id));
    }

  }

  continuarCompra() {
    // Navegamos al checkout enviando la función y las butacas por estado
    this.continuandoCompra.set(true);
    this.router.navigate(['/compra/checkout'], {
      state: {
        funcionId: this.funcionId,
        butacasIds: this.misSeleccionadas()
      }
    });
  }

  ngOnDestroy() {
    // Si el usuario sale de la página sin comprar, liberamos SUS butacas
    if (!this.continuandoCompra()) {
      this.misSeleccionadas().forEach(id => {
        this.reservasService.liberarButaca(this.funcionId, id);
      });
      this.reservasService.limpiarMisReservas();
    }
  }

import { Injectable, signal, effect, OnDestroy, inject } from '@angular/core';
import { RealtimeChannel } from '@supabase/supabase-js';
import { AuthService } from '../../core/auth/auth.service';
import { SupabaseService } from '../../core/supabase.service';

export interface ButacaOcupada {
    id: string; // ID de la butaca
    estado: 'ocupada' | 'reservada'; // ocupada = ya pagada, reservada = temporal
}

@Injectable({
    providedIn: 'root'
})
export class ReservasService implements OnDestroy {

    private auth = inject(AuthService);
    private supabase = inject(SupabaseService);

    private channel: RealtimeChannel | null = null;

    // Signal reactivo con las butacas no disponibles en este momento
    butacasBloqueadas = signal<ButacaOcupada[]>([]);
    misReservasActivas = signal<string[]>([]);

    private sessionId = this.obtenerSessionId();
    private obtenerSessionId(): string {
        let sid = sessionStorage.getItem('cine_session_id');
        if (!sid) {
            sid = crypto.randomUUID();
            sessionStorage.setItem('cine_session_id', sid);
        }
        return sid;
    }

    /**
     * RF-COM-02: Se suscribe a los cambios en tiempo real de la función actual.
     * Escucha tanto reservas temporales (reservas_butaca) como entradas definitivas (entradas).
     */
    async inicializarTiempoReal(funcionId: string) {
        this.limpiarSuscripcion();
        await this.limpiarReservasVencidas();
        await this.cargarOcupacionInicial(funcionId);

        // Canal de Supabase Realtime para esta función específica
        this.channel = this.supabase.client.channel(`ocupacion_funcion_${funcionId}`)
            // Escuchar nuevas reservas temporales
            .on('postgres_changes', {
                event: '*',
                schema: 'public',
                table: 'reservas_butaca',
                filter: `funcion_id=eq.${funcionId}`
            }, () => {
                this.cargarOcupacionInicial(funcionId);
            })
            // Escuchar compras definitivas
            .on('postgres_changes', {
                event: '*',
                schema: 'public',
                table: 'entradas',
                filter: `funcion_id=eq.${funcionId}`
            }, () => {
                this.cargarOcupacionInicial(funcionId);
            })
            .subscribe();
    }

    private async cargarOcupacionInicial(funcionId: string) {
        // 1. Cargar entradas ya vendidas
        const { data: vendidas } = await this.supabase.client
            .from('entradas')
            .select('butaca_id')
            .eq('funcion_id', funcionId)
            .in('estado', ['valida', 'usada']);

        // 2. Cargar reservas temporales vigentes
        const { data: reservadas } = await this.supabase.client
            .from('reservas_butaca')
            .select('butaca_id, session_id')
            .eq('funcion_id', funcionId)
            .gt('expira_at', new Date().toISOString()); // Solo las que no vencieron

        const bloqueadas: ButacaOcupada[] = [];
        const misReservas: string[] = [];

        vendidas?.forEach(v => bloqueadas.push({ id: v.butaca_id, estado: 'ocupada' }));

        reservadas?.forEach(r => {
            // Si la reserva es de ESTA sesión, no la marcamos como bloqueada para este usuario
            if (r.session_id !== this.sessionId) {
                bloqueadas.push({ id: r.butaca_id, estado: 'reservada' });
            } else {
                misReservas.push(r.butaca_id);
            }
        });

        this.butacasBloqueadas.set(bloqueadas);
        this.misReservasActivas.set(misReservas);
    }

    /**
     * Intenta reservar una butaca temporalmente (10 min).
     * RN-10: La tabla tiene un UNIQUE(funcion_id, butaca_id). Si choca, falla acá.
     */
    async reservarButaca(funcionId: string, butacaId: string): Promise<boolean> {
        const userId = this.auth.perfil()?.id || null;

        const { error } = await this.supabase.client
            .from('reservas_butaca')
            .insert({
                funcion_id: funcionId,
                butaca_id: butacaId,
                user_id: userId,
                session_id: this.sessionId
                // expira_at se calcula automáticamente en la BD por DEFAULT
            });

        if (error) {
            if (error.code === '23505') { // Violación de UNIQUE constraint (chocó con otro)
                console.warn('La butaca acaba de ser tomada por otra persona.');
                return false;
            }
            throw new Error(`Error reservando butaca: ${error.message}`);
        }

        // Si fue exitoso, forzamos la recarga para que el usuario la vea como suya
        await this.cargarOcupacionInicial(funcionId);
        return true;
    }

    async liberarButaca(funcionId: string, butacaId: string) {
        const { data, error } = await this.supabase.client
            .from('reservas_butaca')
            .delete()
            .match({
                funcion_id: funcionId,
                butaca_id: butacaId,
                session_id: this.sessionId
            })
            .select(); // <--- CLAVE: si esto devuelve array vacío, el match falló

        if (error) {
            console.error('Error liberando butaca:', error.message);
        } else if (!data || data.length === 0) {
            console.warn(`Supabase borró 0 filas. Revisa en tu base de datos si la butaca ${butacaId} realmente tiene el session_id: ${this.sessionId}`);
        }

        await this.cargarOcupacionInicial(funcionId);
    }

    /**
     * Limpia filas basura de reservas caducadas en lugar de usar un cron.
     */
    async limpiarReservasVencidas() {
        await this.supabase.client
            .from('reservas_butaca')
            .delete()
            .lt('expira_at', new Date().toISOString());
    }

    async limpiarMisReservas() {
        const { data, error } = await this.supabase.client
            .from('reservas_butaca')
            .delete()
            .eq('session_id', this.sessionId)
            .select(); // Obligamos a que devuelva lo borrado

        if (error) console.error('Error al limpiar sesión:', error.message);
    }

    limpiarSuscripcion() {
        if (this.channel) {
            this.supabase.client.removeChannel(this.channel);
            this.channel = null;
        }
    }

    ngOnDestroy() {
        this.limpiarSuscripcion();
    }
}

export class CuentaDashboardComponent implements OnInit {

  private auth = inject(AuthService);

  procesandoCancelacion = signal(false);

      ]);
      this.perfilDatos.set(perfil);
      this.historial.set(compras);
      this.perfil.set(this.auth.perfil());

    } finally {
      this.cargando.set(false);
    }

    this.estadoModal = 'confirmacion';
    this.mensajeModal = '';
    this.mostrarModalCancelacion = true;
  }

  cerrarModalCancelacion() {
    this.mostrarModalCancelacion = false;
    this.ordenSeleccionada = null;
  }

    if (!this.ordenSeleccionada) return;

    const orden = this.ordenSeleccionada;

    this.procesandoCancelacion.set(true);

            : o
        )
      );

      this.mensajeModal = e.message || 'Ocurrió un error al intentar cancelar la compra.';

      this.estadoModal = 'error';

      this.procesandoCancelacion.set(false);

import { Injectable, inject } from '@angular/core';
import { SupabaseService } from '../../core/supabase.service';
import { AuthService } from '../../core/auth/auth.service';


@Injectable({
  providedIn: 'root'
})
export class CuentaService {
  private supabase = inject(SupabaseService);
  private auth = inject(AuthService);

  async getDatosPerfil() {
    const userId = this.auth.perfil()?.id;
    if (!userId) throw new Error('Usuario no autenticado');

    const [{ data: puntos, error: errPuntos }, { data: creditos, error: errCreditos }] = await Promise.all([
      this.supabase.client.from('puntos_movimientos').select('puntos').eq('user_id', userId),
      this.supabase.client.from('creditos_movimientos').select('monto').eq('user_id', userId),
    ]);

    if (errPuntos) throw errPuntos;
    if (errCreditos) throw errCreditos;

    return {
      saldoPuntos: puntos.reduce((acc, m) => acc + m.puntos, 0),
      saldoCredito: creditos.reduce((acc, m) => acc + Number(m.monto), 0),
    };
  }

  async getHistorialCompras() {
    const userId = this.auth.perfil()?.id;
    const { data, error } = await this.supabase.client
      .from('ordenes')
      .select(`
        id, total, estado, created_at,
        entradas ( id, estado, funciones ( inicio, peliculas(titulo, poster_url) ) )
      `)
      .eq('user_id', userId)
      .order('created_at', { ascending: false });

    if (error) throw error;
    return data;
  }

  /**
   * REGLA DE NEGOCIO: Cancelación hasta 2 horas antes y reintegro en créditos.
   */
  async cancelarOrden(orden: any) {
    const userId = this.auth.perfil()?.id;

    const funcionInicio = new Date(orden.entradas[0]?.funciones?.inicio);
    const ahora = new Date();
    const diferenciaHoras = (funcionInicio.getTime() - ahora.getTime()) / (1000 * 60 * 60);

    if (diferenciaHoras < 2) {
      throw new Error('Solo se puede cancelar hasta 2 horas antes de la función.');
    }
    const { data: puntosDeEstaOrden } = await this.supabase.client
      .from('puntos_movimientos')
      .select('puntos')
      .eq('orden_id', orden.id)
      // Asegúrate de que el string coincida con el tipo que usas al sumar puntos en el checkout
      .eq('tipo', 'acumulacion')
      .maybeSingle();

    // Si la orden le dio puntos, tomamos ese valor. Si no (es vieja o anónima), es 0.
    const puntosARevertir = puntosDeEstaOrden ? puntosDeEstaOrden.puntos : 0;

    if (puntosARevertir > 0) {
      // 2. Obtenemos el saldo actual sumando todo el historial
      const { data: movimientos } = await this.supabase.client
        .from('puntos_movimientos')
        .select('puntos')
        .eq('user_id', userId);

      const saldoActualPuntos = movimientos?.reduce((acc, mov) => acc + (mov.puntos || 0), 0) || 0;

      // 3. Comprobamos si le alcanza
      if (saldoActualPuntos < puntosARevertir) {
        throw new Error(`No puedes cancelar. Tienes ${saldoActualPuntos} puntos, pero necesitas ${puntosARevertir} para devolver los generados por esta compra.`);
      }
    }

    await this.supabase.client.from('ordenes').update({ estado: 'cancelada' }).eq('id', orden.id);
    await this.supabase.client.from('entradas').update({ estado: 'cancelada' }).eq('orden_id', orden.id);

    // B) Acreditar: un INSERT, no un update sobre un valor leído antes.
    const { error: errCredito } = await this.supabase.client.from('creditos_movimientos').insert({
      user_id: userId,
      tipo: 'acreditacion',
      monto: orden.total,
      orden_id: orden.id,
      descripcion: `Crédito por cancelación de orden #${orden.id}`,
    });
    if (errCredito) throw errCredito;

    // RN-07 (del archivo de ambigüedades, A-07): revertir los puntos ganados por esa orden
    const { error: errPuntos } = await this.supabase.client.from('puntos_movimientos').insert({
      user_id: userId,
      tipo: 'ajuste',
      puntos: -puntosARevertir, // mismo criterio que al acreditar: 1 punto por peso
      orden_id: orden.id,
      descripcion: `Reverso de puntos por cancelación de orden #${orden.id}`,
    });
    if (errPuntos) throw errPuntos;

    await this.supabase.client.from('cancelaciones').insert({
      orden_id: orden.id,
      user_id: userId,
      monto_acreditado: orden.total,
    });

    await this.supabase.client.from('audit_log').insert({
      actor_id: userId,
      accion: 'cancelacion_usuario',
      entidad: 'ordenes',
      entidad_id: orden.id,
    });

    const nuevo = await this.getDatosPerfil();
    return nuevo.saldoCredito;
  }

  /**
   * Obtiene las recompensas configuradas por el admin
   */
  async getRecompensas() {
    const { data, error } = await this.supabase.client
      .from('recompensas')
      .select('*')
      .eq('activa', true);

    if (error) throw error;
    return data;
  }

  /**
 * Obtiene las películas que el usuario ya vio (entradas en estado 'usada' o de funciones pasadas)
 * y adjunta la reseña si el usuario ya la calificó.
 */
  async getMisPeliculasVistas() {
    const userId = this.auth.perfil()?.id;
    if (!userId) return [];

    // 1. Obtener entradas válidas/usadas del usuario
    const { data: entradas, error: errEntradas } =
      await this.supabase.client
        .from('entradas')
        .select(`
        id,
        estado,
        ordenes!inner (
          user_id
        ),
        funciones!inner (
          inicio,
          peliculas!inner (
            id,
            titulo,
            poster_url
          )
        )
      `)
        .eq('ordenes.user_id', userId)
        .eq('estado', 'usada');

    if (errEntradas) {
      throw new Error(errEntradas.message);
    }

    // 2. Obtener las reseñas que este usuario ya dejó
    const { data: resenas, error: errResenas } = await this.supabase.client
      .from('resenias')
      .select('*')
      .eq('user_id', userId);

    if (errResenas) throw new Error('Error al cargar reseñas.');

    // 3. Agrupar por película en TypeScript para un historial visual único
    const peliculasMap = new Map<string, any>();

    entradas?.forEach((entrada: any) => {
      const peli = entrada.funciones.peliculas;
      const fecha = entrada.funciones.inicio;

      if (!peliculasMap.has(peli.id)) {
        const miResena = resenas?.find(r => r.pelicula_id === peli.id);
        peliculasMap.set(peli.id, {
          id: peli.id,
          titulo: peli.titulo,
          poster_url: peli.poster_url,
          ultima_fecha: fecha,
          calificacion: miResena?.puntaje || 0,
          comentario: miResena?.comentario || ''
        });
      } else {
        // Si la vio varias veces, guardar la fecha más reciente
        const existente = peliculasMap.get(peli.id);
        if (new Date(fecha) > new Date(existente.ultima_fecha)) {
          existente.ultima_fecha = fecha;
        }
      }
    });

    return Array.from(peliculasMap.values());
  }

  /**
   * Inserta o actualiza la reseña de una película (Upsert).
   */
  async guardarResena(peliculaId: string, puntaje: number, comentario: string) {
    const userId = this.auth.perfil()?.id;
    if (!userId) throw new Error('Debe iniciar sesión para calificar.');

    const { error } = await this.supabase.client
      .from('resenias')
      .upsert({
        user_id: userId,
        pelicula_id: peliculaId,
        puntaje: puntaje,
        comentario: comentario,
        updated_at: new Date().toISOString()
      }, { onConflict: 'user_id, pelicula_id' }); // Requiere restricción UNIQUE en BD

    if (error) throw new Error(`Error al guardar calificación: ${error.message}`);

    // Auditoría
    await this.supabase.client.from('audit_log').insert({
      actor_id: userId,
      accion: 'dejar_resena',
      entidad: 'resenias',
      entidad_id: peliculaId
    });
  }
}

  tipo = signal<TipoValidacion>('entrada');
  codigoManual = '';
  
  procesando = signal(false);
  resultadoExito = signal(false);
  mensajeError = signal<string | null>(null);

  datosCandy = signal<any>(null);

  async validarCodigo() {
    if (!this.codigoManual.trim()) return;
    
    this.procesando.set(true);
    this.resultadoExito.set(false);
    this.mensajeError.set(null);
    this.datosEntrada.set(null);
    this.datosCandy.set(null);

    try {
      if (this.tipo() === 'entrada') {
        const entrada = await this.validacionService.validarEntrada(this.codigoManual.trim());
        this.datosEntrada.set(entrada);
      } else {
        const orden = await this.validacionService.validarRetiroCandy(this.codigoManual.trim());
        this.datosCandy.set(orden);
      }
      this.resultadoExito.set(true);
      this.codigoManual = ''; // Limpiar para el siguiente escaneo
    } catch (e: any) {
      this.mensajeError.set(e.message);
    } finally {
      this.procesando.set(false);
    }
  }

import { Injectable, inject } from '@angular/core';
import { AuthService } from '../../core/auth/auth.service';
import { SupabaseService } from '../../core/supabase.service';


@Injectable({
  providedIn: 'root'
})
export class ValidacionService {
  private auth = inject(AuthService);
  private supabase = inject(SupabaseService);

  /**
   * Valida el acceso a la sala escaneando el QR de una entrada.
   * Verifica que no haya sido usada antes y registra quién la validó.
   */
  async validarEntrada(codigoQr: string) {
    const empleadoId = this.auth.perfil()?.id;

    // 1. Buscar la entrada
    const { data: entrada, error: errBusqueda } = await this.supabase.client
      .from('entradas')
      .select('*, funciones(pelicula_id, inicio, formato, idioma)')
      .eq('codigo_qr', codigoQr)
      .single();

    if (errBusqueda || !entrada) throw new Error('Código QR no encontrado o inválido.');

    // 2. Verificar estado (QR de un solo uso)
    if (entrada.estado === 'usada') throw new Error('Esta entrada YA FUE VALIDADA anteriormente.');
    if (entrada.estado === 'cancelada') throw new Error('Esta entrada fue cancelada y no es válida.');

    // 3. Quemar la entrada (marcar como usada)
    const { error: errUpdate } = await this.supabase.client
      .from('entradas')
      .update({
        estado: 'usada',
        validada_at: new Date().toISOString(),
        validada_por: empleadoId
      })
      .eq('id', entrada.id);

    if (errUpdate) throw new Error('Error de conexión al validar la entrada.');

    // 4. Registrar auditoría (quién validó)
    await this.registrarAuditoria('validar_entrada', 'entradas', entrada.id);

    return entrada;
  }

  /**
   * Valida el retiro de productos del Candy Bar usando el código de la orden.
   */
  async validarRetiroCandy(codigoCandy: string) {
    const empleadoId = this.auth.perfil()?.id;

    const { data: orden, error: errBusqueda } = await this.supabase.client
      .from('ordenes')
      .select('id, candy_estado, orden_productos(*)')
      .eq('codigo_candy', codigoCandy)
      .single();

    if (errBusqueda || !orden) throw new Error('Código de Candy Bar no encontrado.');
    if (orden.candy_estado === 'entregado') throw new Error('Este pedido de Candy Bar YA FUE ENTREGADO.');
    if (orden.candy_estado === 'cancelado') throw new Error('Este pedido fue cancelado.');

    const { error: errUpdate } = await this.supabase.client
      .from('ordenes')
      .update({
        candy_estado: 'entregado',
        candy_entregado_at: new Date().toISOString(),
        candy_entregado_por: empleadoId
      })
      .eq('id', orden.id);

    if (errUpdate) throw new Error('Error al confirmar la entrega del Candy Bar.');

    await this.registrarAuditoria('entregar_candy', 'ordenes', orden.id);

    return orden;
  }

  private async registrarAuditoria(accion: string, entidad: string, entidadId: string) {
    const perfil = this.auth.perfil();
    await this.supabase.client.from('audit_log').insert({
      actor_id: perfil?.id,
      actor_rol: perfil?.rol,
      accion: accion,
      entidad: entidad,
      entidad_id: entidadId
    });
  }

import { Component, computed, effect, input, output, signal } from '@angular/core';

const NOMBRES_MES = [
  'Enero', 'Febrero', 'Marzo', 'Abril', 'Mayo', 'Junio',
  'Julio', 'Agosto', 'Septiembre', 'Octubre', 'Noviembre', 'Diciembre',
];

/** Cantidad de días de un mes dado (1-12), respetando años bisiestos. */
function diasEnMes(anio: number, mes: number): number {
  return new Date(anio, mes, 0).getDate();
}

@Component({
  imports: [],
  selector: 'app-fecha-hora-select',
  styleUrl: './fecha-hora-select.component.css',
  templateUrl: './fecha-hora-select.component.html',
})
export class FechaHoraSelectComponent {

  /** Fecha inicial opcional. Si no se pasa, arranca en "ahora". */
  valorInicial = input<Date | undefined>(undefined);

  /** Rango de años a ofrecer hacia adelante (programación de funciones a futuro). */
  aniosHaciaAdelante = input(2);

  /** Emite la fecha completa cada vez que el usuario cambia cualquier select. */
  valorCambio = output<Date>();

  // --- Estado interno, con signals ---
  private readonly ahora = new Date();
  mostrarHora = input(true);

  fechaMinima = input<Date | undefined>(undefined);
  fechaMaxima = input<Date | undefined>(undefined);

  dia = signal(this.ahora.getDate());
  mes = signal(this.ahora.getMonth() + 1); // 1-12, más natural para mostrar que 0-11
  anio = signal(this.ahora.getFullYear());
  hora = signal(this.ahora.getHours());

  meses = NOMBRES_MES.map((nombre, i) => ({ valor: i + 1, nombre }));
  horas = Array.from({ length: 24 }, (_, i) => i);
  minutos = [0, 15, 30, 45]; // cambiá a [0..59] si necesitás precisión al minuto exacto
  minuto = signal(this.redondearMinuto(this.ahora.getMinutes()));

  // Los días del mes dependen del mes/año elegidos (ej. febrero tiene 28 o 29)
  diasDelMes = computed(() => {
    const anio = this.anio();
    const mes = this.mes();

    const fechaMin = this.fechaMinima();
    const fechaMax = this.fechaMaxima();

    let diaMin = 1;
    let diaMax = diasEnMes(anio, mes);

    // Límite inferior
    if (
      fechaMin &&
      anio === fechaMin.getFullYear() &&
      mes === fechaMin.getMonth() + 1
    ) {
      diaMin = fechaMin.getDate();
    }

    // Límite superior
    if (
      fechaMax &&
      anio === fechaMax.getFullYear() &&
      mes === fechaMax.getMonth() + 1
    ) {
      diaMax = fechaMax.getDate();
    }

    return Array.from(
      { length: diaMax - diaMin + 1 },
      (_, i) => diaMin + i
    );
  });

  anios = computed(() => {
    const base = this.ahora.getFullYear();

    const minimo = this.fechaMinima()?.getFullYear() ?? base;
    const maximo =
      this.fechaMaxima()?.getFullYear() ??
      base + this.aniosHaciaAdelante();

    return Array.from(
      { length: maximo - minimo + 1 },
      (_, i) => minimo + i
    );
  });
  mesesDisponibles = computed(() => {
    const anio = this.anio();

    return this.meses.filter(m => {
      if (this.fechaMinima() &&
        anio === this.fechaMinima()!.getFullYear() &&
        m.valor < this.fechaMinima()!.getMonth() + 1) {
        return false;
      }

      if (this.fechaMaxima() &&
        anio === this.fechaMaxima()!.getFullYear() &&
        m.valor > this.fechaMaxima()!.getMonth() + 1) {
        return false;
      }

      return true;
    });
  });



  /** La fecha final, recalculada automáticamente cuando cambia cualquier parte. */
  fecha = computed(() => new Date(this.anio(), this.mes() - 1, this.dia(), this.hora(), this.minuto()));

  constructor() {
    // Si el padre pasa un valor inicial, sembramos los signals con él.
    effect(() => {
      const inicial = this.valorInicial();
      if (inicial) {
        this.dia.set(inicial.getDate());
        this.mes.set(inicial.getMonth() + 1);
        this.anio.set(inicial.getFullYear());
        this.hora.set(inicial.getHours());
        this.minuto.set(this.redondearMinuto(inicial.getMinutes()));
      }
    });

    // Cada vez que la fecha calculada cambia, se la avisamos al padre.
    effect(() => this.valorCambio.emit(this.fecha()));
  }

  onDiaChange(e: Event) {
    this.dia.set(Number((e.target as HTMLSelectElement).value));
  }

  onMesChange(e: Event) {
    const nuevoMes = Number(
      (e.target as HTMLSelectElement).value
    );

    this.mes.set(nuevoMes);

    const diasPermitidos = this.diasDelMes();

    if (!diasPermitidos.includes(this.dia())) {
      this.dia.set(diasPermitidos[0]);
    }
  }

  onAnioChange(e: Event) {
  const nuevoAnio = Number(
    (e.target as HTMLSelectElement).value
  );

  this.anio.set(nuevoAnio);

  const diasPermitidos = this.diasDelMes();

  if (!diasPermitidos.includes(this.dia())) {
    this.dia.set(diasPermitidos[0]);
  }
}

  onHoraChange(e: Event) {
    this.hora.set(Number((e.target as HTMLSelectElement).value));
  }

  onMinutoChange(e: Event) {
    this.minuto.set(Number((e.target as HTMLSelectElement).value));
  }

  private redondearMinuto(m: number): number {
    // Lleva el minuto actual al valor de "minutos" más cercano hacia abajo.
    return this.minutos.reduce((mejor, actual) => (actual <= m ? actual : mejor), 0);
  }

}

import { Component, computed, effect, inject, OnInit, signal } from '@angular/core';
import { AuthService } from '../../../core/auth/auth.service';
import { CuentaService } from '../../../features/cuenta/cuenta.service';
import { Router, RouterLink, RouterLinkActive } from '@angular/router';
import { SlicePipe } from '@angular/common';
import { ModalComponent } from '../../utilidades/modal/modal.component';
import { LoginComponent } from '../../../features/auth/login/login.component';
import { RegistroComponent } from '../../../features/auth/registro/registro.component';

@Component({
  imports: [RouterLink, RouterLinkActive, SlicePipe, ModalComponent, LoginComponent, RegistroComponent],
  selector: 'app-navbar',
  styleUrl: './navbar.component.css',
  templateUrl: './navbar.component.html',
})
export class NavbarComponent implements OnInit {

  private auth = inject(AuthService);
  private cuentaService = inject(CuentaService);
  private router = inject(Router);

  // Signals para manejar la UI
  saldos = signal({ saldoPuntos: 0, saldoCredito: 0 });
  userEmail = signal<string>('');
  modalActivo: 'login' | 'register' | null = null;


  userInitials = signal<string>('U');
  mostrarMenu = signal<boolean>(false);


  abrirModal(tipo: 'login' | 'register') {
    this.modalActivo = tipo;
  }

  cerrarModal() {
    this.modalActivo = null;
  }

  constructor(){
    effect(() => {
    this.cargarSaldos();
  });
  }

  async ngOnInit() {

    if (this.auth.estaAutenticado()) {
      const perfil = this.auth.perfil();

      if (perfil?.email) {
        this.userEmail.set(perfil.email);
        this.userInitials.set(perfil.email.substring(0, 2));
      }

      try {
        this.cargarSaldos();
      } catch (error) {
        console.error('Error al cargar saldos en el Navbar', error);
      }
    }
  }

  async cargarSaldos() {
    const datosLedger = await this.cuentaService.getDatosPerfil();
    this.saldos.set({
      saldoPuntos: datosLedger.saldoPuntos || 0,
      saldoCredito: datosLedger.saldoCredito || 0
    });
  }

  async cerrarSesion() {
    await this.auth.cerrarSesion();
    this.router.navigate(['/']);
    window.location.reload();
  }

}

import { Component, ElementRef, input, viewChild } from '@angular/core';

@Component({
  imports: [],
  selector: 'app-slider',
  styleUrl: './slider.component.css',
  templateUrl: './slider.component.html',
})
export class SliderComponent {

  slider = viewChild<ElementRef<HTMLDivElement>>('slider');

  desplazamiento = input<number>(0.8);

  deslizar(direccion: 'izquierda' | 'derecha') {

    const elemento = this.slider()?.nativeElement;

    if (!elemento) return;

    const cantidad = elemento.clientWidth * this.desplazamiento();

    elemento.scrollBy({
      left: direccion === 'derecha'
        ? cantidad
        : -cantidad,

      behavior: 'smooth'
    });
  }
  
}
